Блок схеми и C код

Символите са стандартни: овал – начало/край, успоредник – въвеждане/извеждане, правоъгълник – действие, ромб – условие. „Да“ излиза от дясното, а „Не“ – от лявото ъгълче на ромба. Връщането в потока става върху линията, а всяка стрелка влиза в блока отгоре. Означенията са като в C: = присвояване, == сравнение, % остатък, / целочислено деление.

Група 1

Задача 1 – сума на числата, които са > 10 или четни

#include <stdio.h>

int main() {
    int n, x, sum = 0;
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) {
        scanf("%d", &x);
        if (x > 10 || x % 2 == 0) {
            sum = sum + x;
        }
    }
    printf("%d", sum);
    return 0;
}

Задача 3 – число с най-голяма сума на цифрите

#include <stdio.h>

int main() {
    int n, num, copyNum, digitSum;
    int maxDigitSum = 0, maxNum = 0;
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) {
        scanf("%d", &num);
        copyNum = num;
        digitSum = 0;
        while (copyNum > 0) {
            digitSum = digitSum + copyNum % 10;
            copyNum = copyNum / 10;
        }
        if (digitSum > maxDigitSum) {
            maxDigitSum = digitSum;
            maxNum = num;
        }
    }
    printf("%d", maxNum);
    return 0;
}

Задача 2 – разлика между най-голямата и най-малката цифра

#include <stdio.h>

int main() {
    int n, digit;
    int max = 0, min = 9;
    scanf("%d", &n);
    while (n > 0) {
        digit = n % 10;
        if (digit > max) {
            max = digit;
        }
        if (digit < min) {
            min = digit;
        }
        n = n / 10;
    }
    printf("%d", max - min);
    return 0;
}

Задача 3 – сума на четните числа, съдържащи цифрата 7

Решение 1 – с !has7 в условието на цикъла

#include <stdio.h>

int main() {
    int n, num, copyNum, has7;
    int sum = 0;
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) {
        scanf("%d", &num);
        if (num % 2 == 0) {
            copyNum = num;
            has7 = 0;
            while (copyNum > 0 && !has7) {
                if (copyNum % 10 == 7) {
                    has7 = 1;
                }
                copyNum = copyNum / 10;
            }
            if (has7) {
                sum = sum + num;
            }
        }
    }
    printf("%d", sum);
    return 0;
}

Решение 2 – с break

#include <stdio.h>

int main() {
    int n, num, copyNum, has7;
    int sum = 0;
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) {
        scanf("%d", &num);
        if (num % 2 == 0) {
            copyNum = num;
            has7 = 0;
            while (copyNum > 0) {
                if (copyNum % 10 == 7) {
                    has7 = 1;
                    break;
                }
                copyNum = copyNum / 10;
            }
            if (has7) {
                sum = sum + num;
            }
        }
    }
    printf("%d", sum);
    return 0;
}

Бонус – числа с точно 1, 2 и 3 крайни нули

#include <stdio.h>

int main() {
    int a, b;
    int one = 0, two = 0, three = 0;
    scanf("%d %d", &a, &b);
    for (int x = a; x <= b; x++) {
        if (x % 10000 == 0) {
            continue; // 4 или повече нули – не се брои
        } else if (x % 1000 == 0) {
            three = three + 1;
        } else if (x % 100 == 0) {
            two = two + 1;
        } else if (x % 10 == 0) {
            one = one + 1;
        }
    }
    printf("%d %d %d", one, two, three);
    return 0;
}
